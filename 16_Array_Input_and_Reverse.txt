// Array Input & Reverse

import java.util.*;

public class Main {
    PSVM(String[] args) {
        Scanner sc = new Scanner(System.in);
        int size = sc.nextInt();
        int[] arr = new int[size];
        for (int i = 0; i < size; i++) {
            arr[i] = sc.nextInt();
        }
        SOP(arr);
        // # Reverse
        for (int i = size - 1; i < size; i--) {
            SOP(arr[i] + " ");
        }
    }
}
