// Palindrome Array (Two Pointers While Loop)

public class Main {
    PSVM {
        int[] arr = {1, 2, 3, 2, 1};
        boolean isPalindrome = true;
        int left = 0;
        int right = arr.length - 1;
        while (left < right) {
            if (arr[left] != arr[right]) {
                isPalindrome = false;
                break;
            }
            left++;
            right--;
        }
        if (isPalindrome) {
            SOP("Palindrome");
        } else {
            SOP("Not Palindrome");
        }
    }
}
